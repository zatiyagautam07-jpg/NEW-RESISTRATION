<script>
"use strict";

const EVENTS = [
  {name:"Technical Symposium",description:"Technical presentations and engineering projects."},
  {name:"Hackathon",description:"Build creative software solutions with a team."},
  {name:"Project Exhibition",description:"Showcase engineering projects and prototypes."},
  {name:"Cultural Fest",description:"Student performances and campus activities."},
  {name:"Sports",description:"Cricket, football, volleyball and badminton."},
  {name:"NCC",description:"Drills, parades, discipline training and community service."}
];

const $ = id => document.getElementById(id);
const DB_KEY = "collegeEventSystemV2";
let scanner = null;
let scannerRunning = false;
let lastScannedText = "";
let lastScannedAt = 0;

function emptyDB(){
  return {students:[],attendance:[]};
}

function getDB(){
  try{
    const data = JSON.parse(localStorage.getItem(DB_KEY));
    if(!data || !Array.isArray(data.students) || !Array.isArray(data.attendance)){
      return emptyDB();
    }
    return data;
  }catch(e){
    return emptyDB();
  }
}

function saveDB(db){
  try{
    localStorage.setItem(DB_KEY,JSON.stringify(db));
    return true;
  }catch(e){
    alert("Unable to save data. Browser storage may be full or disabled.");
    return false;
  }
}

function escapeHTML(value){
  return String(value ?? "").replace(/[&<>"']/g,c=>({
    "&":"&amp;","<":"&lt;",">":"&gt;",'"':"&quot;","'":"&#39;"
  })[c]);
}

function notice(id,text,type="success"){
  const el=$(id);
  el.textContent=text;
  el.className="notice "+type;
  el.style.display="block";
}

function hideNotice(id){
  $(id).style.display="none";
}

function getStudent(enrollment){
  const key=String(enrollment ?? "").trim().toLowerCase();
  return getDB().students.find(s=>s.enrollment.toLowerCase()===key);
}

function showPage(id){
  if(id!=="scanner" && scannerRunning) stopScanner();

  document.querySelectorAll(".page").forEach(p=>p.classList.remove("active"));
  if(!$(id)) return;
  $(id).classList.add("active");

  document.querySelectorAll("[data-page]").forEach(b=>{
    b.classList.toggle("active",b.dataset.page===id);
  });

  $("navLinks").classList.remove("open");

  if(id==="events") renderEvents();
  if(id==="status") refreshStatus();
  if(id==="qr") generateQR();
  if(id==="scanner") renderRecentAttendance();
  if(id==="admin") renderAdmin();
}

document.querySelectorAll("[data-page]").forEach(b=>{
  b.addEventListener("click",()=>showPage(b.dataset.page));
});

$("menuToggle").addEventListener("click",()=>{
  $("navLinks").classList.toggle("open");
});

function initEventOptions(){
  $("eventGrid").innerHTML=EVENTS.map(e=>`
    <article class="event-card">
      <h3>${escapeHTML(e.name)}</h3>
      <p>${escapeHTML(e.description)}</p>
      <button class="btn" data-register="${escapeHTML(e.name)}">Register for Event</button>
    </article>
  `).join("");

  $("scanEvent").innerHTML=
    '<option value="">Select Event</option>'+
    EVENTS.map(e=>`<option value="${escapeHTML(e.name)}">${escapeHTML(e.name)}</option>`).join("");

  $("eventGrid").querySelectorAll("[data-register]").forEach(b=>{
    b.addEventListener("click",()=>registerEvent(b.dataset.register));
  });
}

$("studentForm").addEventListener("submit",e=>{
  e.preventDefault();
  if(!$("studentForm").reportValidity()) return;

  const enrollment=$("enrollment").value.trim();
  const db=getDB();
  const existing=db.students.find(s=>s.enrollment.toLowerCase()===enrollment.toLowerCase());

  const student={
    fullName:$("fullName").value.trim(),
    phone:$("phone").value.trim(),
    email:$("email").value.trim(),
    college:$("college").value.trim(),
    semester:$("semester").value,
    course:$("course").value,
    enrollment,
    events:existing ? existing.events : [],
    status:"Registered",
    createdAt:existing ? existing.createdAt : new Date().toISOString()
  };

  if(existing){
    const confirmed=confirm("This enrollment number already exists. Update this student's profile?");
    if(!confirmed) return;
    db.students=db.students.map(s=>s.enrollment.toLowerCase()===enrollment.toLowerCase()?student:s);
  }else{
    db.students.push(student);
  }

  if(!saveDB(db)) return;
  notice("profileMessage",existing?"Student profile updated successfully.":"Student profile registered successfully.");
  generateQR();
  refreshStatus();
  renderAdmin();
});

$("studentForm").addEventListener("reset",()=>{
  setTimeout(()=>hideNotice("profileMessage"),0);
});

function registerEvent(name){
  const enrollment=$("enrollment").value.trim();
  const db=getDB();

  let student=db.students.find(s=>s.enrollment.toLowerCase()===enrollment.toLowerCase());

  // Use the saved profile when the form is not currently filled.
  if(!student){
    const typed=prompt("Enter your registered enrollment number:");
    if(!typed){
      notice("eventMessage","Please register a student profile first.","error");
      return;
    }
    student=db.students.find(s=>s.enrollment.toLowerCase()===typed.trim().toLowerCase());
  }

  if(!student){
    notice("eventMessage","Student not found. Register a profile first.","error");
    return;
  }

  if(student.events.includes(name)){
    notice("eventMessage","Already registered for "+name+". Duplicate registration prevented.","error");
    return;
  }

  student.events.push(name);
  if(!saveDB(db)) return;

  notice("eventMessage","Successfully registered "+student.fullName+" for "+name+".");
  renderEvents();
  refreshStatus();
  renderAdmin();
}

function renderEvents(){
  const db=getDB();
  const enrollment=$("enrollment").value.trim();
  const student=db.students.find(s=>s.enrollment.toLowerCase()===enrollment.toLowerCase());

  if(!student){
    $("myEvents").textContent="Enter your registered enrollment number in the profile form to see your events, or register for an event and identify your account.";
    return;
  }

  $("myEvents").innerHTML=student.events.length
    ?student.events.map(name=>`<div class="event-card"><strong>${escapeHTML(name)}</strong><p>Registration: Confirmed</p><p>Attendance: ${db.attendance.some(a=>a.enrollment===student.enrollment && a.event===name)?"Checked in":"Not checked in"}</p></div>`).join("")
    :"No events registered yet.";
}

function refreshStatus(){
  const db=getDB();
  const enrollment=$("enrollment").value.trim();
  const student=db.students.find(s=>s.enrollment.toLowerCase()===enrollment.toLowerCase());

  if(!student){
    $("statusContent").textContent="Enter your enrollment number in the profile form, or register a student profile first.";
    return;
  }

  $("statusContent").innerHTML=`
    <div class="event-card">
      <p><strong>Name:</strong> ${escapeHTML(student.fullName)}</p>
      <p><strong>Enrollment:</strong> ${escapeHTML(student.enrollment)}</p>
      <p><strong>College:</strong> ${escapeHTML(student.college)}</p>
      <p><strong>Course:</strong> ${escapeHTML(student.course)}</p>
      <p><strong>Semester:</strong> ${escapeHTML(student.semester)}</p>
      <p><strong>Status:</strong> ${escapeHTML(student.status)}</p>
      <p><strong>Registered events:</strong> ${student.events.length ? student.events.map(escapeHTML).join(", "):"None"}</p>
      <p><strong>Events attended:</strong> ${db.attendance.filter(a=>a.enrollment===student.enrollment).map(a=>escapeHTML(a.event)).join(", ")||"None"}</p>
    </div>`;
  renderEvents();
}

function generateQR(){
  const student=getStudent($("enrollment").value.trim());
  const container=$("qrcode");
  container.replaceChildren();

  if(!student){
    $("qrMessage").textContent="Register a profile or enter a registered enrollment number first.";
    return;
  }

  if(typeof QRCode==="undefined"){
    $("qrMessage").textContent="QR library unavailable. Check your internet connection.";
    return;
  }

  // QR includes an enrollment identifier, not phone, email, or password.
  const data=JSON.stringify({
    app:"CollegeEventSystem",
    enrollment:student.enrollment
  });

  new QRCode(container,{
    text:data,width:210,height:210,
    colorDark:"#312e81",colorLight:"#ffffff",
    correctLevel:QRCode.CorrectLevel.M
  });

  $("qrMessage").textContent="QR code for "+student.fullName+" ("+student.enrollment+").";
}

function downloadQR(){
  const canvas=$("qrcode").querySelector("canvas");
  const img=$("qrcode").querySelector("img");
  const url=canvas?canvas.toDataURL("image/png"):img?.src;

  if(!url){
    alert("Generate your QR code first.");
    return;
  }

  const a=document.createElement("a");
  a.href=url;
  a.download="student-qr-code.png";
  a.click();
}

function processCheckin(decodedText){
  if(!$("scanEvent").value){
    notice("scanMessage","Select an event before scanning.","error");
    return;
  }

  let payload;
  try{
    payload=JSON.parse(decodedText);
  }catch(e){
    notice("scanMessage","Invalid QR code. Scan a QR code generated by this website.","error");
    return;
  }

  if(payload.app!=="CollegeEventSystem" || !payload.enrollment){
    notice("scanMessage","This QR code is not a recognized student QR code.","error");
    return;
  }

  recordAttendance(payload.enrollment,$("scanEvent").value);
}

function recordAttendance(enrollment,eventName){
  const db=getDB();
  const student=db.students.find(s=>s.enrollment.toLowerCase()===String(enrollment).trim().toLowerCase());

  if(!student){
    notice("scanMessage","Student not found. Register the student profile first.","error");
    return;
  }

  if(!student.events.includes(eventName)){
    notice("scanMessage",student.fullName+" is not registered for "+eventName+". Check-in denied.","error");
    return;
  }

  const duplicate=db.attendance.some(a=>
    a.enrollment.toLowerCase()===student.enrollment.toLowerCase() &&
    a.event===eventName
  );

  if(duplicate){
    notice("scanMessage","Already checked in: "+student.fullName+" for "+eventName+". Duplicate scan prevented.","error");
    renderRecentAttendance();
    return;
  }

  db.attendance.push({
    enrollment:student.enrollment,
    fullName:student.fullName,
    event:eventName,
    checkedInAt:new Date().toISOString()
  });

  if(!saveDB(db)) return;

  notice("scanMessage","Check-in successful! "+student.fullName+" | "+student.enrollment+" | "+eventName);
  renderRecentAttendance();
  renderAdmin();
  refreshStatus();
}

$("manualCheckin").addEventListener("click",()=>{
  const enrollment=$("manualEnrollment").value.trim();
  const eventName=$("scanEvent").value;

  if(!enrollment || !eventName){
    notice("scanMessage","Enter an enrollment number and select an event.","error");
    return;
  }

  recordAttendance(enrollment,eventName);
});

async function startScanner(){
  if(scannerRunning) return;

  if(typeof Html5Qrcode==="undefined"){
    notice("scanMessage","QR scanner library did not load. Check your internet connection.","error");
    return;
  }

  if(!$("scanEvent").value){
    notice("scanMessage","Select an event before starting the scanner.","error");
    return;
  }

  try{
    scanner=new Html5Qrcode("reader");
    await scanner.start(
      {facingMode:"environment"},
      {fps:10,qrbox:{width:240,height:240}},
      decodedText=>{
        const now=Date.now();
        if(decodedText===lastScannedText && now-lastScannedAt<2500) return;
        lastScannedText=decodedText;
        lastScannedAt=now;
        processCheckin(decodedText);
      },
      ()=>{}
    );
    scannerRunning=true;
    notice("scanMessage","Camera started. Point it at a student's QR code.");
  }catch(error){
    scanner=null;
    scannerRunning=false;
    notice("scanMessage","Could not start camera. Allow camera permission and use HTTPS or localhost. You can also use manual check-in.","error");
  }
}

async function stopScanner(){
  if(scanner && scannerRunning){
    try{await scanner.stop();}catch(e){}
    try{scanner.clear();}catch(e){}
  }
  scanner=null;
  scannerRunning=false;
}

$("startScan").addEventListener("click",startScanner);
$("stopScan").addEventListener("click",stopScanner);

function renderRecentAttendance(){
  const records=getDB().attendance.slice().reverse().slice(0,10);

  $("recentAttendance").innerHTML=records.length
    ?tableHTML(["Student","Enrollment","Event","Check-in Time"],
      records.map(a=>[
        a.fullName,a.enrollment,a.event,new Date(a.checkedInAt).toLocaleString()
      ]))
    :"No attendance recorded yet.";
}

function tableHTML(headers,rows){
  if(!rows.length) return "<p>No records found.</p>";

  return "<div class='table-wrap'><table><thead><tr>"+
    headers.map(h=>"<th>"+escapeHTML(h)+"</th>").join("")+
    "</tr></thead><tbody>"+
    rows.map(row=>"<tr>"+row.map(v=>"<td>"+escapeHTML(v)+"</td>").join("")+"</tr>").join("")+
    "</tbody></table></div>";
}

function renderAdmin(){
  const db=getDB();
  const registrations=db.students.reduce((sum,s)=>sum+s.events.length,0);

  $("totalStudents").textContent=db.students.length;
  $("totalRegistrations").textContent=registrations;
  $("totalAttendance").textContent=db.attendance.length;

  const query=$("studentSearch").value.trim().toLowerCase();
  const students=db.students.filter(s=>
    s.fullName.toLowerCase().includes(query) ||
    s.enrollment.toLowerCase().includes(query)
  );

  $("studentTable").innerHTML=tableHTML(
    ["Name","Enrollment","College","Course","Semester","Phone"],
    students.map(s=>[s.fullName,s.enrollment,s.college,s.course,s.semester,s.phone])
  );

  const regRows=[];
  db.students.forEach(s=>s.events.forEach(event=>{
    regRows.push([s.fullName,s.enrollment,event,
      db.attendance.some(a=>a.enrollment===s.enrollment && a.event===event)?"Checked in":"Not checked in"]);
  }));

  $("registrationTable").innerHTML=tableHTML(
    ["Student","Enrollment","Event","Attendance"],regRows
  );

  $("attendanceTable").innerHTML=tableHTML(
    ["Student","Enrollment","Event","Date and Time"],
    db.attendance.slice().reverse().map(a=>[
      a.fullName,a.enrollment,a.event,new Date(a.checkedInAt).toLocaleString()
    ])
  );
}

$("studentSearch").addEventListener("input",renderAdmin);

function exportCSV(type){
  const db=getDB();
  let headers,rows;

  if(type==="students"){
    headers=["Full Name","Enrollment","Phone","Email","College","Semester","Course","Events"];
    rows=db.students.map(s=>[s.fullName,s.enrollment,s.phone,s.email,s.college,s.semester,s.course,s.events.join("; ")]);
  }else if(type==="registrations"){
    headers=["Full Name","Enrollment","Event","Attendance"];
    rows=[];
    db.students.forEach(s=>s.events.forEach(event=>{
      rows.push([s.fullName,s.enrollment,event,
        db.attendance.some(a=>a.enrollment===s.enrollment && a.event===event)?"Checked in":"Not checked in"]);
    }));
  }else{
    headers=["Full Name","Enrollment","Event","Check-in Time"];
    rows=db.attendance.map(a=>[a.fullName,a.enrollment,a.event,new Date(a.checkedInAt).toLocaleString()]);
  }

  const csvCell=value=>'"'+String(value??"").replace(/"/g,'""')+'"';
  const csv=[headers,...rows].map(row=>row.map(csvCell).join(",")).join("\r\n");
  const blob=new Blob(["\uFEFF"+csv],{type:"text/csv;charset=utf-8;"});
  const url=URL.createObjectURL(blob);
  const a=document.createElement("a");
  a.href=url;
  a.download=type+"-report.csv";
  a.click();
  URL.revokeObjectURL(url);
}
